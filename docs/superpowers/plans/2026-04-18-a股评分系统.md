```python
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
from sqlalchemy.dialects.sqlite import insert as sqlite_insert
from backend.config import DATABASE_URL
from backend.models import Base

engine = create_engine(DATABASE_URL, connect_args={"check_same_thread": False})
SessionLocal = sessionmaker(bind=engine, autocommit=False, autoflush=False)

def init_db():
    Base.metadata.create_all(bind=engine)

def get_session():
    return SessionLocal()

def upsert(session, model, data: dict, index_elements: list):
    stmt = sqlite_insert(model).values(**data)
    update_cols = {k: stmt.excluded[k] for k in data if k not in index_elements}
    stmt = stmt.on_conflict_do_update(index_elements=index_elements, set_=update_cols)
    session.execute(stmt)
```

```python
import pywencai
import pandas as pd
import logging

logger = logging.getLogger(__name__)

def query_wencai(loop_ask: str, query_type: str = "stock") -> pd.DataFrame:
    """Execute a pywencai query, return empty DataFrame on failure."""
    try:
        result = pywencai.get(loop_ask=loop_ask, query_type=query_type)
        if result is None or result.empty:
            return pd.DataFrame()
        return result
    except Exception as e:
        logger.error(f"pywencai query failed: {loop_ask[:50]}... Error: {e}")
        return pd.DataFrame()

def normalize_code(code: str) -> str:
    """Normalize stock code to 6-digit string."""
    return str(code).zfill(6).split(".")[0]
```

```python
import json
import logging
from datetime import date
from backend.collectors.base import query_wencai, normalize_code
from backend.database import upsert
from backend.models import Stock

logger = logging.getLogger(__name__)

def get_hot_sector_stocks(session) -> list[str]:
    """获取当日热门板块 Top10 中的股票"""
    df = query_wencai("涨跌幅排名前10的行业板块", query_type="plate")
    if df.empty:
        return []
    # 取板块名称
    sector_col = [c for c in df.columns if "板块" in c or "行业" in c]
    if not sector_col:
        return []
    sectors = df[sector_col[0]].tolist()[:10]

    codes = []
    for sector in sectors:
        sector_df = query_wencai(f"{sector}板块 涨跌幅 股票代码", query_type="stock")
        if not sector_df.empty:
            code_col = [c for c in sector_df.columns if "代码" in c or "code" in c.lower()]
            if code_col:
                top5 = sector_df[code_col[0]].tolist()[:5]
                codes.extend([normalize_code(c) for c in top5])
    return list(set(codes))

def sync_universe(session, watchlist_codes: list[str] = None):
    """同步全量股票池到 stocks 表"""
    today = date.today().isoformat()

    # 1. 沪深300 + 中证500 + 创业板成分股
    index_stocks = {}
    for index_query, tag in [
        ("沪深300成分股 股票代码 股票名称", "hs300"),
        ("中证500成分股 股票代码 股票名称", "zz500"),
        ("创业板指成分股 股票代码 股票名称", "cyb"),
    ]:
        df = query_wencai(index_query)
        if df.empty:
            continue
        code_col = [c for c in df.columns if "代码" in c]
        name_col = [c for c in df.columns if "名称" in c]
        if not code_col or not name_col:
            continue
        for _, row in df.iterrows():
            code = normalize_code(row[code_col[0]])
            name = str(row[name_col[0]])
            if code not in index_stocks:
                index_stocks[code] = {"name": name, "tags": []}
            index_stocks[code]["tags"].append(tag)

    for code, info in index_stocks.items():
        market = "SH" if code.startswith("6") else "SZ"
        upsert(session, Stock, {
            "code": code, "name": info["name"], "market": market,
            "is_watchlist": False, "index_tags": json.dumps(info["tags"]),
        }, ["code"])

    # 2. 热门板块股票
    hot_codes = get_hot_sector_stocks(session)
    for code in hot_codes:
        existing = session.get(Stock, code)
        if not existing:
            market = "SH" if code.startswith("6") else "SZ"
            upsert(session, Stock, {
                "code": code, "name": code, "market": market,
                "is_watchlist": False, "index_tags": "[]",
            }, ["code"])

    # 3. 自选股
    if watchlist_codes:
        for code in watchlist_codes:
            code = normalize_code(code)
            existing = session.get(Stock, code)
            if existing:
                existing.is_watchlist = True
            else:
                market = "SH" if code.startswith("6") else "SZ"
                upsert(session, Stock, {
                    "code": code, "name": code, "market": market,
                    "is_watchlist": True, "index_tags": "[]",
                }, ["code"])

    session.commit()
    total = session.query(Stock).count()
    logger.info(f"Stock universe synced: {total} stocks total")
    return total
```

```python
import logging
from datetime import date
from backend.collectors.base import query_wencai, normalize_code
from backend.database import upsert
from backend.models import DailyData

logger = logging.getLogger(__name__)

TECHNICAL_QUERY = (
    "股票代码,股票名称,最新价,"
    "5日均线,13日均线,30日均线,"
    "MACD的DIF值,MACD的DEA值,MACD的MACD值,"
    "RSI(14),"
    "KDJ的K值,KDJ的D值,KDJ的J值,"
    "布林线上轨,布林线中轨,布林线下轨,"
    "量比"
)

FIELD_MAP = {
    "close": ["最新价", "收盘价", "close"],
    "ma5":   ["5日均线", "MA5"],
    "ma13":  ["13日均线", "MA13"],
    "ma30":  ["30日均线", "MA30"],
    "macd_dif": ["MACD的DIF值", "DIF"],
    "macd_dea": ["MACD的DEA值", "DEA"],
    "macd_bar": ["MACD的MACD值", "MACD"],
    "rsi14":    ["RSI(14)", "RSI14", "RSI"],
    "kdj_k":    ["KDJ的K值", "K"],
    "kdj_d":    ["KDJ的D值", "D"],
    "kdj_j":    ["KDJ的J值", "J"],
    "boll_upper": ["布林线上轨", "BOLL_UPPER"],
    "boll_mid":   ["布林线中轨", "BOLL_MID"],
    "boll_lower": ["布林线下轨", "BOLL_LOWER"],
    "volume_ratio": ["量比"],
}

def _find_col(df, candidates: list[str]) -> str | None:
    for c in candidates:
        if c in df.columns:
            return c
    return None

def collect_technical(session, target_codes: set[str] = None):
    """Collect technical indicators for all stocks via pywencai."""
    today = date.today().isoformat()
    df = query_wencai(TECHNICAL_QUERY)
    if df.empty:
        logger.warning("Technical data query returned empty")
        return 0

    code_col = _find_col(df, ["股票代码", "代码", "code"])
    if not code_col:
        logger.error("Cannot find code column in technical data")
        return 0

    count = 0
    for _, row in df.iterrows():
        code = normalize_code(row[code_col])
        if target_codes and code not in target_codes:
            continue
        record = {"code": code, "date": today}
        for field, candidates in FIELD_MAP.items():
            col = _find_col(df, candidates)
            record[field] = float(row[col]) if col and row[col] is not None else None
        try:
            upsert(session, DailyData, record, ["code", "date"])
            count += 1
        except Exception as e:
            logger.error(f"Failed to upsert technical data for {code}: {e}")

    session.commit()
    logger.info(f"Technical data collected: {count} stocks")
    return count
```

```python
def clamp(value: float, min_val: float, max_val: float) -> float:
    return max(min_val, min(max_val, value))

def percentile_score(value, all_values: list, higher_is_better: bool = True) -> int:
    """Return 0-100 score based on percentile rank within universe."""
    valid = [v for v in all_values if v is not None]
    if not valid or value is None:
        return 50
    rank = sum(1 for v in valid if v <= value) / len(valid)
    score = rank if higher_is_better else (1 - rank)
    return int(clamp(score * 100, 0, 100))

def safe_float(value, default=None) -> float | None:
    try:
        return float(value)
    except (TypeError, ValueError):
        return default
```

```python
from backend.scorers.technical import TechnicalScorer
from backend.scorers.capital import CapitalScorer
from backend.scorers.fundamental import FundamentalScorer
from backend.scorers.news import NewsScorer
from backend.scorers.market_heat import HeatScorer
from backend.database import upsert
from backend.models import Score, Strategy

logger = logging.getLogger(__name__)

UNIVERSE_FIELDS = {
    "main_inflow_today", "main_inflow_5d", "super_large_inflow",
    "pe", "pb", "roe", "profit_growth_yoy", "change_pct", "turnover_rate",
}

class ScoreEngine:
    def __init__(self):
        self.technical = TechnicalScorer()
        self.capital = CapitalScorer()
        self.fundamental = FundamentalScorer()
        self.news = NewsScorer()
        self.heat = HeatScorer()

    def _build_universe_stats(self, records: list[dict]) -> dict:
        stats = {f: [] for f in UNIVERSE_FIELDS}
        for r in records:
            for f in UNIVERSE_FIELDS:
                v = r.get(f)
                if v is not None:
                    stats[f].append(float(v))
        return stats

    def score_stock(self, data: dict, universe: list[dict], strategy) -> dict:
        stats = self._build_universe_stats(universe)
        t = self.technical.score(data)
        c = self.capital.score(data, stats)
        f = self.fundamental.score(data, stats)
        n = self.news.score(data)
        h = self.heat.score(data, stats)
        total = (
            t * strategy.technical_weight +
            c * strategy.capital_weight +
            f * strategy.fundamental_weight +
            n * strategy.news_weight +
            h * strategy.heat_weight
        )
        return {
            "technical_score": round(t, 2),
            "capital_score":   round(c, 2),
            "fundamental_score": round(f, 2),
            "news_score":      round(n, 2),
            "heat_score":      round(h, 2),
            "total_score":     round(total, 2),
        }

    def run(self, session):
        """Score all stocks in daily_data for today across all strategies."""
        from backend.models import DailyData
        today = date.today().isoformat()
        records = session.query(DailyData).filter(DailyData.date == today).all()
        if not records:
            logger.warning(f"No daily data found for {today}")
            return 0

        universe = [r.__dict__ for r in records]
        strategies = session.query(Strategy).all()
        scored = 0

        for strategy in strategies:
            for record in records:
                data = record.__dict__
                scores = self.score_stock(data, universe, strategy)
                score_record = {
                    "code": record.code, "date": today, "strategy": strategy.name,
                    **scores,
                }
                upsert(session, Score, score_record, ["code", "date", "strategy"])
                scored += 1

        session.commit()
        logger.info(f"Scoring complete: {scored} records written")
        return scored
```

```python
        from datetime import date

        code = code.zfill(6)

        score = (
            session.query(Score)
            .filter(Score.code == code, Score.date == today, Score.strategy == strategy)
            .first()
        )
        stock = session.get(Stock, code)
        daily = (
            session.query(DailyData)
            .filter(DailyData.code == code, DailyData.date == today)
            .first()
        )

        return {

            "name": stock.name if stock else code,
            "scores": {
                "total": score.total_score,
                "technical": score.technical_score,
                "capital": score.capital_score,
                "fundamental": score.fundamental_score,
                "news": score.news_score,
                "heat": score.heat_score,
            },
            "raw": {k: v for k, v in daily.__dict__.items() if not k.startswith("_")} if daily else {},

    finally:
```

```python
import threading
import logging
from datetime import datetime
from fastapi import APIRouter
from backend.database import get_session
from backend.collectors.universe import sync_universe
from backend.collectors.technical import collect_technical
from backend.collectors.capital import collect_capital
from backend.collectors.fundamental import collect_fundamental
from backend.collectors.news import collect_news
from backend.collectors.market_heat import collect_market_heat
from backend.engine import ScoreEngine

router = APIRouter(prefix="/api/trigger", tags=["trigger"])
logger = logging.getLogger(__name__)

_status = {"running": False, "last_run": None, "last_result": None}

def _run_pipeline():
    _status["running"] = True
    session = get_session()
    try:
        sync_universe(session)
        from backend.models import Stock
        codes = {s.code for s in session.query(Stock).all()}
        collect_technical(session, codes)
        collect_capital(session, codes)
        collect_fundamental(session, codes)
        collect_news(session, codes)
        collect_market_heat(session, codes)
        engine = ScoreEngine()
        count = engine.run(session)
        _status["last_result"] = f"Scored {count} records"
        logger.info(_status["last_result"])
    except Exception as e:
        _status["last_result"] = f"Error: {e}"
        logger.error(e)
    finally:
        session.close()
        _status["running"] = False
        _status["last_run"] = datetime.now().isoformat()

@router.post("/collect")
def trigger_collect():
    if _status["running"]:
        return {"status": "already_running"}
    thread = threading.Thread(target=_run_pipeline, daemon=True)
    thread.start()
    return {"status": "started"}

@router.get("/status")
def get_status():
    return _status
```
